cmake_minimum_required(VERSION 3.22)
project(px4_isaac_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(SHARED_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(ISAAC_ROOT "${SHARED_ROOT}/.venv-isaac/lib/python3.12/site-packages/isaacsim" CACHE PATH "Isaac runtime")
set(NATIVE_SDK "${SHARED_ROOT}/.deps/native-sdk" CACHE PATH "Native SDK headers")
set(PX4_ROOT "${SHARED_ROOT}/.deps/PX4-Autopilot" CACHE PATH "PX4 checkout")
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
add_library(flight_core STATIC native/src/flight_core.cpp)
target_include_directories(flight_core PUBLIC native/include PRIVATE "${PX4_ROOT}/src/lib/events/libevents/libs/cpp/parse" "${PX4_ROOT}/build/px4_sitl_default/mavlink")
target_compile_options(flight_core PRIVATE -Wall -Wextra -Wno-address-of-packed-member)
add_library(kit_sdk INTERFACE)
target_include_directories(kit_sdk SYSTEM INTERFACE "${ISAAC_ROOT}/kit/dev/include" "${ISAAC_ROOT}/kit/dev/fabric/include" "${ISAAC_ROOT}/kit/dev/gsl/include" "${ISAAC_ROOT}/kit/dev/ogn/include" "${NATIVE_SDK}/usd/include" /usr/include/python3.12 "${NATIVE_SDK}/PhysX/omni/ovruntime/include" "${NATIVE_SDK}/PhysX/omni/schema/source")
target_compile_definitions(kit_sdk INTERFACE _GLIBCXX_USE_CXX11_ABI=1 TBB_SUPPRESS_DEPRECATED_MESSAGES=1)
target_compile_options(kit_sdk INTERFACE -Wno-deprecated-declarations)
file(GLOB USD_LIB_ROOT "${ISAAC_ROOT}/extscache/omni.usd.libs-*/bin")
file(GLOB USD_CORE_LIB "${ISAAC_ROOT}/extscache/omni.usd.core-*/bin/libomni.usd.so")
file(GLOB PHYSICS_TOOLS_LIB "${ISAAC_ROOT}/extscache/omni.usd.schema.physx-*/bin/libphysicsSchemaTools.so")
set(USD_LIBS "${USD_CORE_LIB}")
foreach(component usd usdGeom usdPhysics usdShade usdRender usdLux usdUtils sdf tf gf vt arch)
    list(APPEND USD_LIBS "${USD_LIB_ROOT}/libusd_${component}.so")
endforeach()
target_link_libraries(kit_sdk INTERFACE ${USD_LIBS} "${ISAAC_ROOT}/kit/libcarb.so" dl Threads::Threads)
function(add_kit_plugin target extension)
    add_library(${target} SHARED ${ARGN})
    target_link_libraries(${target} PRIVATE kit_sdk)
    target_include_directories(${target} PRIVATE native/include)
    target_compile_options(${target} PRIVATE -fvisibility=hidden)
    set_target_properties(${target} PROPERTIES OUTPUT_NAME "${extension}.plugin" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/extensions/${extension}/bin")
endfunction()
add_kit_plugin(flight_plugin px4.isaac.flight native/src/flight_plugin.cpp)
target_link_libraries(flight_plugin PRIVATE flight_core "${PHYSICS_TOOLS_LIB}")
add_executable(isaac_native native/src/isaac_native.cpp)
target_link_libraries(isaac_native PRIVATE kit_sdk flight_core)
target_compile_definitions(isaac_native PRIVATE ISAAC_KIT_PATH="${ISAAC_ROOT}/kit")
set_target_properties(isaac_native PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng)
add_library(gst_camera STATIC native/src/gst_camera.cpp)
target_include_directories(gst_camera PUBLIC native/include)
target_link_libraries(gst_camera PUBLIC PkgConfig::GST Threads::Threads)
add_library(terrain_core STATIC native/src/terrain.cpp)
target_include_directories(terrain_core PUBLIC native/include)
target_link_libraries(terrain_core PUBLIC PkgConfig::CURL PkgConfig::PNG Threads::Threads)
add_kit_plugin(terrain_plugin px4.isaac.terrain native/src/terrain_plugin.cpp)
target_link_libraries(terrain_plugin PRIVATE terrain_core)
add_kit_plugin(camera_plugin px4.isaac.camera native/src/camera_plugin.cpp)
target_link_libraries(camera_plugin PRIVATE gst_camera)
add_executable(px4_launcher native/src/px4_launcher.cpp)
target_include_directories(px4_launcher PRIVATE native/include)
set_target_properties(px4_launcher PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
