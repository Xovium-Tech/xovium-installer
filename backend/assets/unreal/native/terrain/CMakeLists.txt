cmake_minimum_required(VERSION 3.16)
project(xovium_terrain LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_package(JPEG REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
set(NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/..")
file(GLOB TERRAIN "${NATIVE}/vendor/terrain/src/core/*.cc")
add_library(xovium_terrain SHARED terrain.cpp terrain_image.cpp ${TERRAIN})
target_include_directories(xovium_terrain PRIVATE "${NATIVE}/include" "${NATIVE}/vendor/terrain/include" ${OpenCV_INCLUDE_DIRS} ${JPEG_INCLUDE_DIRS})
target_link_libraries(xovium_terrain PRIVATE Threads::Threads ${CMAKE_DL_LIBS} CURL::libcurl ${OpenCV_LIBS})
target_compile_options(xovium_terrain PRIVATE -Wall -Wextra -Wpedantic)
# Unreal exports a different libjpeg ABI; decode in a private linker namespace.
target_compile_definitions(xovium_terrain PRIVATE XOVIUM_PRIVATE_JPEG=1 XOVIUM_JPEG_LIBRARY="${JPEG_LIBRARY_RELEASE}")
set_target_properties(xovium_terrain PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
install(TARGETS xovium_terrain LIBRARY DESTINATION .)
