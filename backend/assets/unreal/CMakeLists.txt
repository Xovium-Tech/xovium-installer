cmake_minimum_required(VERSION 3.16)
project(ue_px4_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAVLINK_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/mavlink/include" CACHE PATH "Generated MAVLink C headers")
if(NOT EXISTS "${MAVLINK_INCLUDE_DIR}/common/mavlink.h")
  message(FATAL_ERROR "Run scripts/generate_mavlink.sh, or set -DMAVLINK_INCLUDE_DIR=.../px4_sitl_default/mavlink")
endif()
add_library(ue_px4_bridge native/src/mavlink_link.cpp)
target_include_directories(ue_px4_bridge PUBLIC native/include)
target_include_directories(ue_px4_bridge SYSTEM PRIVATE "${MAVLINK_INCLUDE_DIR}")
target_compile_options(ue_px4_bridge PRIVATE -Wall -Wextra -Wpedantic)
