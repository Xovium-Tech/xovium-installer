cmake_minimum_required(VERSION 3.16)
project(xovium_sih LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAVLINK_INCLUDE_DIR "" CACHE PATH "Shared PX4 generated MAVLink headers")
if(NOT EXISTS "${MAVLINK_INCLUDE_DIR}/common/mavlink.h")
  message(FATAL_ERROR "Set MAVLINK_INCLUDE_DIR to PX4 build/px4_sitl_default/mavlink")
endif()
find_package(Threads REQUIRED)
add_library(xovium_sih SHARED sih.cpp)
target_include_directories(xovium_sih PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(xovium_sih SYSTEM PUBLIC "${MAVLINK_INCLUDE_DIR}")
target_link_libraries(xovium_sih PRIVATE Threads::Threads)
target_compile_options(xovium_sih PRIVATE -Wall -Wextra -Wpedantic)

# The lightweight rover FDM also drives FlightGear's external FDM input. It
# shares the sensor/transport and streamed elevation implementations with Isaac.
set(FLIGHT_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/native" CACHE PATH "Shared native sensor and terrain source directory")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
if(NOT EXISTS "${FLIGHT_CORE_DIR}/src/flight_core.cpp" OR NOT EXISTS "${FLIGHT_CORE_DIR}/src/terrain.cpp")
  message(FATAL_ERROR "FLIGHT_CORE_DIR must contain native/include and native/src flight_core and terrain sources")
endif()
if(NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set JSON_INCLUDE_DIR to PX4 src/lib/events/libevents/libs/cpp/parse")
endif()
find_package(CURL REQUIRED)
find_package(PNG REQUIRED)
add_library(xovium_rover_core STATIC rover_physics.cpp "${FLIGHT_CORE_DIR}/src/flight_core.cpp" "${FLIGHT_CORE_DIR}/src/terrain.cpp")
target_include_directories(xovium_rover_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${FLIGHT_CORE_DIR}/include")
target_include_directories(xovium_rover_core SYSTEM PUBLIC "${MAVLINK_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}")
target_link_libraries(xovium_rover_core PUBLIC Threads::Threads CURL::libcurl PNG::PNG)
target_compile_options(xovium_rover_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(xovium_rover rover_main.cpp)
target_link_libraries(xovium_rover PRIVATE xovium_rover_core)
