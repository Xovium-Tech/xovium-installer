cmake_minimum_required(VERSION 3.22)
project(xovium_installer VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
set(DEPS "${CMAKE_SOURCE_DIR}/.build/deps")
if(NOT EXISTS "${DEPS}/raylib/CMakeLists.txt")
  message(FATAL_ERROR "Run ./install.sh --build-only to prepare pinned GUI dependencies")
endif()
add_subdirectory("${DEPS}/raylib" "${CMAKE_BINARY_DIR}/raylib" EXCLUDE_FROM_ALL)
add_library(imgui STATIC "${DEPS}/imgui/imgui.cpp" "${DEPS}/imgui/imgui_draw.cpp" "${DEPS}/imgui/imgui_tables.cpp" "${DEPS}/imgui/imgui_widgets.cpp")
target_include_directories(imgui PUBLIC "${DEPS}/imgui")
add_library(rlImGui STATIC "${DEPS}/rlImGui/rlImGui.cpp")
target_include_directories(rlImGui PUBLIC "${DEPS}/rlImGui")
target_link_libraries(rlImGui PUBLIC raylib imgui)
add_executable(xovium-installer app/main.cpp app/process.cpp app/auth.cpp)
add_executable(xovium-askpass app/askpass_main.cpp app/auth.cpp)
target_compile_options(xovium-askpass PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(xovium-installer PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(xovium-installer PRIVATE rlImGui)
